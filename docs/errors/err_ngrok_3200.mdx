input int      MA_Period = 14;       // Moving Average Period
input double   LotSize   = 0.10;     // Lot size
input double   StopLoss  = 200;      // Stop Loss in points
input double   TakeProfit= 400;      // Take Profit in points

int handleMA;

//+------------------------------------------------------------------+
//| Expert initialization                                            |
//+------------------------------------------------------------------+
int OnInit()
  {
   handleMA = iMA(_Symbol, PERIOD_M15, MA_Period, 0, MODE_SMA, PRICE_CLOSE);
   if(handleMA == INVALID_HANDLE)
     {
      Print("Error creating MA indicator");
      return INIT_FAILED;
     }
   return INIT_SUCCEEDED;
  }

//+------------------------------------------------------------------+
//| Expert deinitialization                                          |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
  {
   IndicatorRelease(handleMA);
  }

//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
  {
   double ma[];
   if(CopyBuffer(handleMA,0,0,2,ma) < 2) return;

   double lastMA = ma[1];
   double prevMA = ma[0];

   double lastPrice = iClose(_Symbol, PERIOD_M15, 1);

   // Check open positions
   int total = PositionsTotal();
   bool hasPosition = false;
   for(int i=0; i<total; i++)
     {
      if(PositionGetSymbol(i) == _Symbol)
         hasPosition = true;
     }

   if(!hasPosition)
     {
      if(lastPrice > lastMA) // BUY
        {
         OpenTrade(ORDER_TYPE_BUY);
        }
      else if(lastPrice < lastMA) // SELL
        {
         OpenTrade(ORDER_TYPE_SELL);
        }
     }
  }
